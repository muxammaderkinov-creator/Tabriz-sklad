import sqlite3 import tkinter as tk from tkinter import ttk, messagebox class TabrizSkladApp: def __init__(self, root): self.root = root self.root.title("Tabriz sklad - Учёт мебельного склада") self.root.geometry("1100x700") # Инициализация базы данных self.db_init() # Создание стилей и вкладок self.notebook = ttk.Notebook(self.root) self.notebook.pack(fill=tk.BOTH, expand=True) self.create_sales_tab() # Вкладка быстрых продаж self.create_products_tab() # Вкладка складского учёта self.create_expenses_tab() # Вкладка расходов self.create_clients_tab() # Вкладка клиентов # Строка состояния и подпись в правом нижнем углу self.create_footer() def db_init(self): """Подключение к локальной SQL базе данных SQLite""" self.conn = sqlite3.connect("tabriz_sklad.db") self.cursor = self.conn.cursor() # Создание базовых таблиц self.cursor.executescript(""" CREATE TABLE IF NOT EXISTS products ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, category TEXT, price REAL, stock INT, is_hot INT ); CREATE TABLE IF NOT EXISTS expenses ( id INTEGER PRIMARY KEY AUTOINCREMENT, title TEXT, amount REAL, date TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); """) self.conn.commit() def create_sales_tab(self): """Эргономичная панель быстрых и горячих продаж""" tab = ttk.Frame(self.notebook) self.notebook.add(tab, text="🔥 Горячие продажи / Касса") left_frame = ttk.LabelFrame(tab, text="Товары и Быстрый выбор") left_frame.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=10, pady=10) right_frame = ttk.LabelFrame(tab, text="Чек и Итого") right_frame.pack(side=tk.RIGHT, fill=tk.BOTH, expand=False, padx=10, pady=10) # Быстрые кнопки товаров (Горячие продажи) hot_btn1 = tk.Button(left_frame, text="ЛДСП 18мм (Белый)\n15 000 сум", bg="#e1f5fe", height=3, width=20) hot_btn1.grid(row=0, column=0, padx=5, pady=5) hot_btn2 = tk.Button(left_frame, text="Петля с доводчиком\n2 500 сум", bg="#e1f5fe", height=3, width=20) hot_btn2.grid(row=0, column=1, padx=5, pady=5) # Таблица позиций в чеке columns = ("Товар", "Кол-во", "Цена") self.cart_tree = ttk.Treeview(right_frame, columns=columns, show="headings", height=15) for col in columns: self.cart_tree.heading(col, text=col) self.cart_tree.column(col, width=100) self.cart_tree.pack(fill=tk.BOTH, padx=5, pady=5) btn_checkout = tk.Button(right_frame, text="Завершить продажу", bg="#4caf50", fg="white", font=("Arial", 12, "bold")) btn_checkout.pack(fill=tk.X, padx=5, pady=10) def create_products_tab(self): """Учёт товаров, добавление, остатки и стоимость склада""" tab = ttk.Frame(self.notebook) self.notebook.add(tab, text="📦 Склад и Остатки") # Верхняя инфо-панель (Общая стоимость склада) info_frame = ttk.Frame(tab) info_frame.pack(fill=tk.X, padx=10, pady=5) lbl_total_value = tk.Label(info_frame, text="Общая стоимость склада: 0.00 UZS", font=("Arial", 11, "bold"), fg="#1b5e20") lbl_total_value.pack(side=tk.LEFT) btn_add = tk.Button(info_frame, text="+ Добавить товар", bg="#2196f3", fg="white", command=self.add_product_dialog) btn_add.pack(side=tk.RIGHT) # Таблица остатков cols = ("ID", "Наименование", "Категория", "Цена", "Остаток") self.prod_tree = ttk.Treeview(tab, columns=cols, show="headings") for c in cols: self.prod_tree.heading(c, text=c) self.prod_tree.pack(fill=tk.BOTH, expand=True, padx=10, pady=5) def create_expenses_tab(self): """Учёт расходов на складе""" tab = ttk.Frame(self.notebook) self.notebook.add(tab, text="💸 Расходы склада") lbl = ttk.Label(tab, text="Учёт арендной платы, транспортировки и прочих затрат") lbl.pack(anchor=tk.W, padx=10, pady=10) def create_clients_tab(self): """Клиентская база""" tab = ttk.Frame(self.notebook) self.notebook.add(tab, text="👥 Клиенты") lbl = ttk.Label(tab, text="База покупателей и история взаиморасчётов") lbl.pack(anchor=tk.W, padx=10, pady=10) def add_product_dialog(self): """Диалог добавления нового товара""" dialog = tk.Toplevel(self.root) dialog.title("Добавить товар") dialog.geometry("350x300") tk.Label(dialog, text="Название товара:").pack(anchor=tk.W, padx=10, pady=2) entry_name = tk.Entry(dialog) entry_name.pack(fill=tk.X, padx=10, pady=2) tk.Label(dialog, text="Цена продажи:").pack(anchor=tk.W, padx=10, pady=2) entry_price = tk.Entry(dialog) entry_price.pack(fill=tk.X, padx=10, pady=2) tk.Label(dialog, text="Количество:").pack(anchor=tk.W, padx=10, pady=2) entry_qty = tk.Entry(dialog) entry_qty.pack(fill=tk.X, padx=10, pady=2) def save(): name = entry_name.get() price = float(entry_price.get() or 0) qty = int(entry_qty.get() or 0) self.cursor.execute("INSERT INTO products (name, price, stock) VALUES (?, ?, ?)", (name, price, qty)) self.conn.commit() messagebox.showinfo("Успешно", "Товар успешно добавлен!") dialog.destroy() btn_save = tk.Button(dialog, text="Сохранить", bg="#4caf50", fg="white", command=save) btn_save.pack(pady=15) def create_footer(self): """Нижняя панель с постоянной надписью в правом углу""" footer = tk.Frame(self.root, bg="#eceff1", height=25) footer.pack(fill=tk.X, side=tk.BOTTOM) lbl_status = tk.Label(footer, text="Система готова к работе", bg="#eceff1", fg="#546e7a", font=("Arial", 9)) lbl_status.pack(side=tk.LEFT, padx=10) # Обязательный брендинг в правом нижнем углу lbl_academy = tk.Label( footer, text="MUSTAFAYEV ACADEMY", bg="#eceff1", fg="#0d47a1", font=("Arial", 10, "bold") ) lbl_academy.pack(side=tk.RIGHT, padx=10) if __name__ == "__main__": root = tk.Tk() app = TabrizSkladApp(root) root.mainloop()
