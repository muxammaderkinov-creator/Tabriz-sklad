Таблица категорий (Плитные материалы, фурнитура, кромки) CREATE TABLE IF NOT EXISTS categories ( category_id INTEGER PRIMARY KEY AUTOINCREMENT, category_name VARCHAR(100) NOT NULL UNIQUE ); -- 2. Таблица товаров мебельного склада CREATE TABLE IF NOT EXISTS products ( product_id INTEGER PRIMARY KEY AUTOINCREMENT, product_name VARCHAR(150) NOT NULL, category_id INTEGER NOT NULL, material VARCHAR(100), -- ЛДСП 18мм, МДФ, Алюминий и т.д. purchase_price DECIMAL(12, 2) NOT NULL DEFAULT 0.00, -- Закупочная стоимость selling_price DECIMAL(12, 2) NOT NULL DEFAULT 0.00, -- Цена продажи stock_quantity INT NOT NULL DEFAULT 0, -- Остаток на складе photo_path TEXT, -- Путь или URL к фото товара is_hot_sale BOOLEAN DEFAULT 0, -- 1 = Горячая продажа (быстрый доступ) FOREIGN KEY (category_id) REFERENCES categories(category_id) ); -- 3. Таблица клиентской базы CREATE TABLE IF NOT EXISTS clients ( client_id INTEGER PRIMARY KEY AUTOINCREMENT, full_name VARCHAR(150) NOT NULL, phone_number VARCHAR(30), debt_amount DECIMAL(12, 2) DEFAULT 0.00 -- Задолженность клиента ); -- 4. Таблица учёта расходов склада CREATE TABLE IF NOT EXISTS warehouse_expenses ( expense_id INTEGER PRIMARY KEY AUTOINCREMENT, title VARCHAR(150) NOT NULL, -- Аренда, Свет, Логистика, ЗП amount DECIMAL(12, 2) NOT NULL, expense_date DATETIME DEFAULT CURRENT_TIMESTAMP ); -- 5. Таблица продаж (Чеки) CREATE TABLE IF NOT EXISTS sales ( sale_id INTEGER PRIMARY KEY AUTOINCREMENT, client_id INTEGER, total_amount DECIMAL(12, 2) NOT NULL DEFAULT 0.00, sale_date DATETIME DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (client_id) REFERENCES clients(client_id) ); -- 6. Детализация продаж (Позиции чека) CREATE TABLE IF NOT EXISTS sale_items ( item_id INTEGER PRIMARY KEY AUTOINCREMENT, sale_id INTEGER NOT NULL, product_id INTEGER NOT NULL, quantity INT NOT NULL, price_per_unit DECIMAL(12, 2) NOT NULL, FOREIGN KEY (sale_id) REFERENCES sales(sale_id), FOREIGN KEY (product_id) REFERENCES products(product_id) ); -- 7.
