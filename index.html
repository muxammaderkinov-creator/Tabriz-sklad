-- Таблица категорий (например: ЛДСП, Фурнитура, Готовая мебель, Кромки) CREATE TABLE IF NOT EXISTS categories ( id INTEGER PRIMARY KEY AUTOINCREMENT, name VARCHAR(100) NOT NULL UNIQUE ); -- Таблица товаров мебельного склада CREATE TABLE IF NOT EXISTS products ( id INTEGER PRIMARY KEY AUTOINCREMENT, barcode VARCHAR(50) UNIQUE, name VARCHAR(150) NOT NULL, category_id INTEGER, material VARCHAR(100), -- Например: ЛДСП 18мм, МДФ, Металл purchase_price DECIMAL(10, 2), -- Себестоимость / закупка selling_price DECIMAL(10, 2), -- Цена продажи stock_quantity INT DEFAULT 0, -- Остаток на складе photo_path TEXT, -- Путь к фото товара или ссылка is_hot BOOLEAN DEFAULT 0, -- Флаг "Горячие продажи" (быстрый выбор) FOREIGN KEY (category_id) REFERENCES categories(id) ); -- Таблица клиентов (Клиентская база) CREATE TABLE IF NOT EXISTS clients ( id INTEGER PRIMARY KEY AUTOINCREMENT, full_name VARCHAR(150) NOT NULL, phone VARCHAR(30), discount_percent INT DEFAULT 0, debt DECIMAL(10, 2) DEFAULT 0.00 -- Задолженность клиента ); -- Таблица продаж CREATE TABLE IF NOT EXISTS sales ( id INTEGER PRIMARY KEY AUTOINCREMENT, client_id INTEGER, total_amount DECIMAL(10, 2) NOT NULL, sale_date DATETIME DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (client_id) REFERENCES clients(id) ); -- Позиции в чеке / продаже CREATE TABLE IF NOT EXISTS sale_items ( id INTEGER PRIMARY KEY AUTOINCREMENT, sale_id INTEGER, product_id INTEGER, quantity INT NOT NULL, price_per_unit DECIMAL(10, 2) NOT NULL, FOREIGN KEY (sale_id) REFERENCES sales(id), FOREIGN KEY (product_id) REFERENCES products(id) ); -- Учёт расходов склада (Аренда, Свет, Зарплата, Доставка) CREATE TABLE IF NOT EXISTS warehouse_expenses ( id INTEGER PRIMARY KEY AUTOINCREMENT, expense_title VARCHAR(150) NOT NULL, category VARCHAR(100), amount DECIMAL(10, 2) NOT NULL, expense_date DATETIME DEFAULT CURRENT_TIMESTAMP, notes TEXT );
